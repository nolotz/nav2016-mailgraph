    PROCEDURE CreateMessage@1(SenderName@1000 : Text;SenderAddress@1001 : Text;Recipients@1002 : Text;Subject@1003 : Text;Body@1004 : Text;HtmlFormatted@1005 : Boolean);
    VAR
      SMTPMailSetup@1006 : Record 409;
      SenderMailbox@1007 : Text;
    BEGIN
      // Same address validation as Codeunit 400 (Codeunit 9520). A blank sender is accepted here when a fixed
      // Graph Sender Mailbox is configured, because the mailbox then becomes the From address.
      IF Recipients <> '' THEN
        MailManagement.CheckValidEmailAddresses(Recipients);
      IF SenderAddress <> '' THEN
        MailManagement.CheckValidEmailAddresses(SenderAddress);

      SMTPMailSetup.GET;
      SenderMailbox := SMTPMailSetup."Graph Sender Mailbox";
      IF (SenderMailbox = '') AND (SenderAddress = '') THEN
        ERROR(SenderMissingErr);

      CLEAR(GraphMessage);
      SendResult := '';
      // Clear error instead of NAV's generic DotNet instantiation error when NavMailGraph.dll is missing.
      IF NOT CANLOADTYPE(GraphMessage) THEN
        ERROR(AddInMissingErr);
      GraphMessage := GraphMessage.GraphMailMessage;
      GraphMessage.FromName := SenderName;
      IF SenderMailbox <> '' THEN BEGIN
        // Fixed mailbox: always send as the mailbox (no Send As needed). The caller's address goes to Reply-To.
        GraphMessage.FromAddress := SenderMailbox;
        IF SMTPMailSetup."Graph Reply-To Caller" AND (SenderAddress <> '') THEN
          IF LOWERCASE(SenderAddress) <> LOWERCASE(SenderMailbox) THEN
            GraphMessage.AddReplyTo(SenderAddress);
      END ELSE
        // No fixed mailbox: the caller's From address is the mailbox used in /users/<mailbox>/sendMail.
        GraphMessage.FromAddress := SenderAddress;

      IF Recipients <> '' THEN
        GraphMessage.AddRecipients(Recipients);
      GraphMessage.Subject := Subject;
      GraphMessage.Body := Body;
      GraphMessage.HtmlFormatted := HtmlFormatted;
    END;

    PROCEDURE AddAttachment@2(Attachment@1000 : Text;FileName@1001 : Text);
    BEGIN
      // Mirrors Codeunit 400: blank path is ignored, a missing server file is an error, the display name is
      // cleaned with Codeunit 419 and ';' is removed. The add-in reads the file at Send time.
      IF Attachment = '' THEN
        EXIT;
      CheckMessageCreated;
      IF NOT EXISTS(Attachment) THEN
        ERROR(AttachmentNotFoundErr,Attachment);
      FileName := FileManagement.StripNotsupportChrInFileName(FileName);
      FileName := DELCHR(FileName,'=',';');
      GraphMessage.AddAttachment(Attachment,FileName);
    END;

    PROCEDURE TrySend@10() : Boolean;
    BEGIN
      // Same contract as Codeunit 400.TrySend: never raises an error, returns SendResult = ''.
      // Configuration errors (missing tenant, secret, add-in ...) are caught by TryDoSend and end up in SendResult,
      // so Codeunit 9520 (HideSMTPError) and GetLastSendMailErrorText behave exactly like with SMTP.
      SendResult := '';
      CLEARLASTERROR;
      IF NOT TryDoSend THEN BEGIN
        SendResult := GETLASTERRORTEXT;
        IF SendResult = '' THEN
          SendResult := UnknownErr;
      END;
      CLEAR(GraphMessage);
      EXIT(SendResult = '');
    END;

    PROCEDURE SetClientSecret@23(VAR SMTPMailSetup@1000 : Record 409;Secret@1001 : Text);
    VAR
      ServicePassword@1002 : Record 1261;
    BEGIN
      // Same pattern as Table 1260 "Bank Data Conv. Service Setup".SavePassword: the secret lives in Table 1261
      // "Service Password" (encrypted by Codeunit 1266 when an encryption key exists); Table 409 only keeps the key.
      // The caller saves SMTPMailSetup (Page 409 does this as part of normal record handling).
      IF Secret = '' THEN BEGIN
        DeleteClientSecret(SMTPMailSetup);
        EXIT;
      END;
      IF ISNULLGUID(SMTPMailSetup."Graph Secret Key") OR NOT ServicePassword.GET(SMTPMailSetup."Graph Secret Key") THEN BEGIN
        ServicePassword.SavePassword(Secret);
        ServicePassword.INSERT(TRUE);
        SMTPMailSetup."Graph Secret Key" := ServicePassword.Key;
      END ELSE BEGIN
        ServicePassword.SavePassword(Secret);
        ServicePassword.MODIFY;
      END;
      IF GUIALLOWED AND NOT ENCRYPTIONKEYEXISTS THEN
        MESSAGE(SecretNotEncryptedMsg);
    END;

    LOCAL PROCEDURE CreateSender@32(VAR SMTPMailSetup@1000 : Record 409);
    BEGIN
      // Maps the Table 409 Graph fields to NavMailGraph.GraphMailSender.
      IF NOT CANLOADTYPE(GraphSender) THEN
        ERROR(AddInMissingErr);
      SMTPMailSetup.TESTFIELD("Graph Tenant ID");
      SMTPMailSetup.TESTFIELD("Graph Client ID");

      CLEAR(GraphSender);
      GraphSender := GraphSender.GraphMailSender(SMTPMailSetup."Graph Tenant ID",SMTPMailSetup."Graph Client ID");

      CASE SMTPMailSetup."Graph Auth Method" OF
        SMTPMailSetup."Graph Auth Method"::"Client Secret":
          BEGIN
            IF NOT HasClientSecret(SMTPMailSetup) THEN
              ERROR(SecretMissingErr);
            GraphSender.UseClientSecret(GetClientSecret(SMTPMailSetup));
          END;
        SMTPMailSetup."Graph Auth Method"::Certificate:
          BEGIN
            SMTPMailSetup.TESTFIELD("Graph Cert. Thumbprint");
            GraphSender.UseCertificateThumbprint(SMTPMailSetup."Graph Cert. Thumbprint");
          END;
      END;

      CASE SMTPMailSetup."Graph Token Library" OF
        SMTPMailSetup."Graph Token Library"::"Built-in":
          GraphSender.TokenLibrary := 'BuiltIn';
        SMTPMailSetup."Graph Token Library"::MSAL:
          GraphSender.TokenLibrary := 'Msal';
      END;

      CASE SMTPMailSetup."Graph Cloud" OF
        SMTPMailSetup."Graph Cloud"::"Global":
          GraphSender.Cloud := 'Global';
        SMTPMailSetup."Graph Cloud"::"US Government":
          GraphSender.Cloud := 'USGov';
        SMTPMailSetup."Graph Cloud"::"US Government DoD":
          GraphSender.Cloud := 'USGovDoD';
        SMTPMailSetup."Graph Cloud"::China:
          GraphSender.Cloud := 'China';
      END;

      GraphSender.SenderMailbox := SMTPMailSetup."Graph Sender Mailbox";
      GraphSender.SaveToSentItems := SMTPMailSetup."Graph Save to Sent Items";
      GraphSender.UseDraftForLargeMessages := SMTPMailSetup."Graph Large Messages";
      GraphSender.LogFilePath := SMTPMailSetup."Graph Log File Path";
      IF SMTPMailSetup."Graph Link Threshold (MB)" > 0 THEN
        ApplyLinkStorage(SMTPMailSetup);
    END;

          BEGIN
            SMTPMailSetup.TESTFIELD("Graph Cert. Thumbprint");

      END;

        SMTPMailSetup."Graph Cloud"::"Global":

        SMTPMailSetup."Graph Cloud"::"US Government":

    LOCAL PROCEDURE ApplyLinkStorage@35(VAR SMTPMailSetup@1000 : Record 409);
    VAR
      StorageProfile@1003 : Record 50403;
      ThresholdBytes@1001 : BigInteger;
      ValidMinutes@1002 : Integer;
    BEGIN
      // NMG2.00: attachments >= "Graph Link Threshold (MB)" are uploaded to the object storage profile and replaced
      // by a time-limited download link in the body. Read-only on the database (runs inside TryDoSend).
      SMTPMailSetup.TESTFIELD("Graph Link Storage Profile");
      ObjectStorageMgt.CreateClient(SMTPMailSetup."Graph Link Storage Profile",LinkStorageClient);
      ThresholdBytes := SMTPMailSetup."Graph Link Threshold (MB)";
      ThresholdBytes := ThresholdBytes * 1048576;
      ValidMinutes := SMTPMailSetup."Graph Link Validity (Days)" * 1440;
      IF ValidMinutes <= 0 THEN
        ValidMinutes := 7 * 1440;
      // Amazon S3 / S3-compatible: pre-signed URLs (SigV4) are valid for at most 7 days; a longer validity would
      // make every send with a link fail, so it is limited here.
      IF StorageProfile.GET(SMTPMailSetup."Graph Link Storage Profile") THEN
        IF (StorageProfile.Provider = StorageProfile.Provider::"Amazon S3") AND (ValidMinutes > 7 * 1440) THEN
          ValidMinutes := 7 * 1440;
      GraphSender.UseLinkStorage(LinkStorageClient,ThresholdBytes,ValidMinutes,'');
    END;
