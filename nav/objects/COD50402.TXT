    PROCEDURE CreateClient@1(ProfileCode@1000 : Code[20];VAR NewStorageClient@1001 : DotNet "'NavMailGraph, Version=1.0.0.0, Culture=neutral, PublicKeyToken=null'.NavMailGraph.ObjectStorageClient");
    VAR
      StorageProfile@1002 : Record 50403;
      MegaByte@1003 : BigInteger;
      Region@1004 : Text;
    BEGIN
      // Maps Table 50403 to NavMailGraph.ObjectStorageClient. Raises an error for configuration problems only;
      // the client itself never throws (operations return '' or an error text).
      IF NOT CANLOADTYPE(NewStorageClient) THEN
        ERROR(AddInMissingErr);
      StorageProfile.GET(ProfileCode);
      CLEAR(NewStorageClient);
      NewStorageClient := NewStorageClient.ObjectStorageClient;

      CASE StorageProfile.Provider OF
        StorageProfile.Provider::"Azure Blob Shared Key":
          BEGIN
            StorageProfile.TESTFIELD("Account Name");
            StorageProfile.TESTFIELD("Container Name");
            CheckSecret(StorageProfile);
            NewStorageClient.UseAzureBlobSharedKey(
              StorageProfile."Account Name",GetSecret(StorageProfile),StorageProfile."Container Name");
          END;
        StorageProfile.Provider::"Azure Blob SAS URL":
          BEGIN
            // The secret is the container SAS URL (https://account.blob.core.windows.net/container?sv=...).
            CheckSecret(StorageProfile);
            NewStorageClient.UseAzureBlobSasUrl(GetSecret(StorageProfile));
          END;
        StorageProfile.Provider::"Azure Blob Entra ID":
          BEGIN
            // Uses the Graph app registration of Table 409; the app needs "Storage Blob Data Contributor".
            StorageProfile.TESTFIELD("Account Name");
            StorageProfile.TESTFIELD("Container Name");
            GraphMail.CreateGraphConnection(GraphConnection);
            NewStorageClient.UseAzureBlobEntra(GraphConnection,StorageProfile."Account Name",StorageProfile."Container Name");
          END;
        StorageProfile.Provider::"Amazon S3":
          BEGIN
            StorageProfile.TESTFIELD("Container Name");
            StorageProfile.TESTFIELD("Access Key ID");
            CheckSecret(StorageProfile);
            Region := StorageProfile.Region;
            IF Region = '' THEN
              Region := 'us-east-1';
            NewStorageClient.UseS3(
              StorageProfile."Service URL",Region,StorageProfile."Container Name",StorageProfile."Access Key ID",
              GetSecret(StorageProfile));
            CASE StorageProfile."S3 Addressing" OF
              StorageProfile."S3 Addressing"::"Path Style":
                NewStorageClient.S3PathStyle := TRUE;
              StorageProfile."S3 Addressing"::"Virtual Hosted Style":
                NewStorageClient.S3PathStyle := FALSE;
            END;
          END;
      END;

      // Options are set after Use...; they only change defaults.
      IF (StorageProfile."Service URL" <> '') AND (StorageProfile.Provider <> StorageProfile.Provider::"Amazon S3") THEN
        NewStorageClient.ServiceUrl := StorageProfile."Service URL";
      IF StorageProfile."Blob Endpoint Suffix" <> '' THEN
        NewStorageClient.BlobEndpointSuffix := StorageProfile."Blob Endpoint Suffix";
      IF StorageProfile."Azure API Version" <> '' THEN
        NewStorageClient.AzureApiVersion := StorageProfile."Azure API Version";
      NewStorageClient.KeyPrefix := StorageProfile."Key Prefix";
      MegaByte := 1048576;
      IF StorageProfile."Multipart Threshold (MB)" > 0 THEN
        NewStorageClient.MultipartThresholdBytes := StorageProfile."Multipart Threshold (MB)" * MegaByte;
      IF StorageProfile."Part Size (MB)" > 0 THEN
        NewStorageClient.PartSizeBytes := StorageProfile."Part Size (MB)" * 1048576;
      IF StorageProfile."Timeout (sec.)" > 0 THEN
        NewStorageClient.TimeoutSeconds := StorageProfile."Timeout (sec.)";
      NewStorageClient.MaxRetries := StorageProfile."Max. Retries";
      NewStorageClient.LogFilePath := StorageProfile."Log File Path";
    END;

    PROCEDURE SetSecret@21(VAR StorageProfile@1000 : Record 50403;Secret@1001 : Text);
    VAR
      ServicePassword@1002 : Record 1261;
    BEGIN
      // Same pattern as Table 1260.SavePassword: the secret lives in Table 1261 "Service Password",
      // encrypted when the database has an encryption key. The caller saves StorageProfile.
      IF Secret = '' THEN BEGIN
        DeleteSecret(StorageProfile);
        EXIT;
      END;
      IF ISNULLGUID(StorageProfile."Secret Key") OR NOT ServicePassword.GET(StorageProfile."Secret Key") THEN BEGIN
        ServicePassword.SavePassword(Secret);
        ServicePassword.INSERT(TRUE);
        StorageProfile."Secret Key" := ServicePassword.Key;
      END ELSE BEGIN
        ServicePassword.SavePassword(Secret);
        ServicePassword.MODIFY;
      END;
      IF GUIALLOWED AND NOT ENCRYPTIONKEYEXISTS THEN
        MESSAGE(SecretNotEncryptedMsg);
    END;
